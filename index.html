4. BUBBLE SORT O(N²)

Algoritma pengurutan harga tiket/hotel dengan perbandingan elemen bersebelahan dan optimasi flag swapped.

bubbleSort.js 
for (let i = 0; i < n - 1; i++) {
  let swapped = false;
  for (let j = 0; j < n - i - 1; j++) {
    if (daftarAktif[j].harga > daftarAktif[j + 1].harga) {
      let temp = daftarAktif[j];
      daftarAktif[j] = daftarAktif[j + 1];
      daftarAktif[j + 1] = temp;
      swapped = true;
    }
  }
  if (!swapped) break;
}
